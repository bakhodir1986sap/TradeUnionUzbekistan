SELECT AAA.ProfsoyzNaimenovanie , CCC.Org_Count , AAA.FEDERATION_AMT , AAA.OBLSOVPROF_AMT
 FROM
(SELECT St.ProfsoyzNaimenovanie , ecp.OBL_DEPARTMENT_BASE , SUM(FEDERATION_AMT) as FEDERATION_AMT, SUM(OBLSOVPROF_AMT) as  OBLSOVPROF_AMT
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
INNER JOIN KasabaKIAT_N1.dbo.tbku_StrokaProfsoyza St
on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
GROUP BY St.ProfsoyzNaimenovanie, ecp.OBL_DEPARTMENT_BASE) As AAA
INNER JOIN
(SELECT BBB.OBL_DEPARTMENT_BASE , COUNT(BBB.INN) as Org_Count FROM
(SELECT ecp.OBL_DEPARTMENT_BASE , ecp.INN 
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
INNER JOIN KasabaKIAT_N1.dbo.tbku_StrokaProfsoyza St
on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
GROUP BY ecp.OBL_DEPARTMENT_BASE , ecp.INN) AS BBB
GROUP BY BBB.OBL_DEPARTMENT_BASE) AS CCC
ON AAA.OBL_DEPARTMENT_BASE = CCC.OBL_DEPARTMENT_BASE
ORDER BY AAA.ProfsoyzNaimenovanie

SELECT AAA.ProfsoyzNaimenovanie , CCC.Org_Count , AAA.FEDERATION_AMT , AAA.OBLSOVPROF_AMT
 FROM
(SELECT St.ProfsoyzNaimenovanie , ecp.OBL_DEPARTMENT_BASE , SUM(FEDERATION_AMT) as FEDERATION_AMT, SUM(OBLSOVPROF_AMT) as  OBLSOVPROF_AMT

on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
AND ecp.BRANCH_CODE is null 
GROUP BY St.ProfsoyzNaimenovanie, ecp.OBL_DEPARTMENT_BASE) As AAA
INNER JOIN
(SELECT BBB.OBL_DEPARTMENT_BASE , COUNT(BBB.INN) as Org_Count FROM
(SELECT ecp.OBL_DEPARTMENT_BASE , ecp.INN 
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
INNER JOIN KasabaKIAT_N1.dbo.tbku_StrokaProfsoyza St
on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
AND ecp.BRANCH_CODE is null 
GROUP BY ecp.OBL_DEPARTMENT_BASE , ecp.INN) AS BBB
GROUP BY BBB.OBL_DEPARTMENT_BASE) AS CCC
ON AAA.OBL_DEPARTMENT_BASE = CCC.OBL_DEPARTMENT_BASE
ORDER BY AAA.ProfsoyzNaimenovanie

SELECT AAA.ProfsoyzNaimenovanie , AAA.BRANCH_NAME , CCC.Org_Count , AAA.FEDERATION_AMT , AAA.OBLSOVPROF_AMT
 FROM
(SELECT St.ProfsoyzNaimenovanie , ecp.OBL_DEPARTMENT_BASE , ISNULL(ecp.BRANCH_NAME, 'Не определен') as BRANCH_NAME, SUM(FEDERATION_AMT) as FEDERATION_AMT, SUM(OBLSOVPROF_AMT) as  OBLSOVPROF_AMT
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
INNER JOIN KasabaKIAT_N1.dbo.tbku_StrokaProfsoyza St
on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
GROUP BY St.ProfsoyzNaimenovanie, ecp.OBL_DEPARTMENT_BASE, ISNULL(ecp.BRANCH_NAME, 'Не определен')) As AAA
INNER JOIN
(SELECT BBB.OBL_DEPARTMENT_BASE , BBB.BRANCH_NAME , COUNT(BBB.INN) as Org_Count FROM
(SELECT ecp.OBL_DEPARTMENT_BASE , ISNULL(ecp.BRANCH_NAME, 'Не определен') as BRANCH_NAME , ecp.INN 
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
INNER JOIN KasabaKIAT_N1.dbo.tbku_StrokaProfsoyza St
on ecp.OBL_DEPARTMENT_BASE = St.MATCH_DEPARTMENTBASE
WHERE YEAR(OPERDATE) = 2020
GROUP BY ecp.OBL_DEPARTMENT_BASE , ISNULL(ecp.BRANCH_NAME, 'Не определен') , ecp.INN) AS BBB
GROUP BY BBB.OBL_DEPARTMENT_BASE , BBB.BRANCH_NAME) AS CCC
ON AAA.OBL_DEPARTMENT_BASE = CCC.OBL_DEPARTMENT_BASE
AND AAA.BRANCH_NAME = CCC.BRANCH_NAME
ORDER BY AAA.ProfsoyzNaimenovanie

SELECT AAA.BRANCH_NAME , CCC.Org_Count , AAA.FEDERATION_AMT , AAA.OBLSOVPROF_AMT
 FROM
(SELECT ISNULL(ecp.BRANCH_NAME, 'Не определен') as BRANCH_NAME, SUM(FEDERATION_AMT) as FEDERATION_AMT, SUM(OBLSOVPROF_AMT) as  OBLSOVPROF_AMT
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
WHERE YEAR(OPERDATE) = 2020
GROUP BY ISNULL(ecp.BRANCH_NAME, 'Не определен')) As AAA
INNER JOIN
(SELECT BBB.BRANCH_NAME , COUNT(BBB.INN) as Org_Count FROM
(SELECT  ISNULL(ecp.BRANCH_NAME, 'Не определен') as BRANCH_NAME , ecp.INN 
FROM KasabaKIAT_N1.dbo.tbbu_ecp_calc ecp
WHERE YEAR(OPERDATE) = 2020
GROUP BY ISNULL(ecp.BRANCH_NAME, 'Не определен') , ecp.INN) AS BBB
GROUP BY BBB.BRANCH_NAME) AS CCC ON
AAA.BRANCH_NAME = CCC.BRANCH_NAME
ORDER BY AAA.BRANCH_NAME
